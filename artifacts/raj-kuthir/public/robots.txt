# Raj Kuthir Homestays
# The guest-facing site is open to crawlers.

User-agent: *
Allow: /

# The owner console (/admin, /sign-in) and the guest arrival pack (/welcome)
# are deliberately NOT disallowed here. Each of them answers with
# <meta name="robots" content="noindex, nofollow"> and an X-Robots-Tag header,
# and a crawler can only obey that if it is allowed to fetch the page. A
# Disallow would stop it reading the noindex, and a blocked URL that is linked
# from somewhere can still be indexed from the link alone. Access to both is
# protected by sign-in and a booking reference, not by this file.

# The API serves data, not pages.
Disallow: /api/

Sitemap: https://rajkuthirhomestays.casa/sitemap.xml
